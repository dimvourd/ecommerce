# run rabbitmq in docker
docker run -d --name rabbitmq -p 5672:5672 -p 15672:15672 -e RABBITMQ_DEFAULT_USER=ecom -e RABBITMQ_DEFAULT_PASS=ecom123 rabbitmq:4-management
docker run -d ^
  --name rabbitmq ^
  -p 5672:5672 ^ -> rabbitMQ message broker (springboot apps connection)
  -p 15672:15672 ^ -> RabbitMQ UI
  -e RABBITMQ_DEFAULT_USER=ecom ^
  -e RABBITMQ_DEFAULT_PASS=ecom123 ^
  rabbitmq:4-management
